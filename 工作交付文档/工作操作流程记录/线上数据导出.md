### mongdb
```bash
# 导出流程
# 进入 mongo 容器
docker exec -it mongo bash

# 导出 hospital_unclassified_records 集合
mongoexport --host mongo --port 27017 \
            --db icd_coding \
            --collection hospital_unclassified_records \
            --username maizi \
            --password maizicare123 \
            --authenticationDatabase admin \
            --out /tmp/hospital_unclassified_records.json \
            --jsonArray
# 从容器里复制出来
docker cp  mongo:/tmp/hospital_unclassified_records.json   /**/hospital_unclassified_records.json

# 导入流程
# 复制到容器里
docker cp  /**/hospital_unclassified_records.json  mongo:/tmp/hospital_unclassified_records.json 
# 进入 mongo 容器
docker exec -it mongo bash

# 导入到目标库（新增）
mongoimport --host mongo --port 27017 \
            --db icd_coding \
            --collection hospital_unclassified_records \
            --username maizi \
            --password maizicare123 \
            --authenticationDatabase admin \
            --file /tmp/hospital_unclassified_records.json \
            --jsonArray \
            --mode upsert
```
